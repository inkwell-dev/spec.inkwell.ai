% Preamble added to pandoc's LaTeX template (pandoc -H). Typesetting rules of
% the report: A4, serif body, sans-serif headings, roman page numbers for the
% front matter and arabic from the general introduction, captions centred with
% "Figure 1.1 – Title", table captions above the table and figure captions below.

% ── Fonts ────────────────────────────────────────────────────────────────────
% Liberation Serif/Sans are metric-compatible with Times New Roman and Arial.
% Glyphs they lack (✓, →, Σ, ≈…) fall back to DejaVu Sans.
\directlua{
  luaotfload.add_fallback("inkwellfallback", {
    "DejaVuSans:mode=node;",
    "DejaVuSansMono:mode=node;",
  })
}
\setmainfont{Liberation Serif}[RawFeature={fallback=inkwellfallback}]
\setsansfont{Liberation Sans}[RawFeature={fallback=inkwellfallback}]
\setmonofont{DejaVu Sans Mono}[Scale=0.85]

% ── Page and paragraphs ──────────────────────────────────────────────────────
\usepackage{setspace}
\setstretch{1.15}
\setlength{\parskip}{4pt plus 1pt}
\usepackage{microtype}
\usepackage{needspace}

% ── Headings ─────────────────────────────────────────────────────────────────
\usepackage{titlesec}
\titleformat{\chapter}[display]
  {\normalfont\sffamily\bfseries\raggedright}
  {\large\MakeUppercase{\chaptertitlename}~\thechapter}
  {6pt}
  {\LARGE}
  [\vspace{2pt}\titlerule]
\titleformat{name=\chapter,numberless}[display]
  {\normalfont\sffamily\bfseries\raggedright}
  {}
  {0pt}
  {\LARGE}
  [\vspace{2pt}\titlerule]
\titlespacing*{\chapter}{0pt}{0pt}{18pt}
\titleformat{\section}{\normalfont\sffamily\Large\bfseries}{\thesection}{0.8em}{}
\titleformat{\subsection}{\normalfont\sffamily\large\bfseries}{\thesubsection}{0.8em}{}
\titleformat{\subsubsection}{\normalfont\sffamily\normalsize\bfseries}{\thesubsubsection}{0.8em}{}
\setcounter{secnumdepth}{2}
\setcounter{tocdepth}{2}

% ── Running header and page numbers ──────────────────────────────────────────
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0.4pt}
\fancyhead[R]{\small\sffamily\nouppercase{\leftmark}}
\fancyfoot[C]{\small\thepage}
\renewcommand{\chaptermark}[1]{\markboth{\ifnum\value{chapter}>0 \ifnum\value{secnumdepth}>-1 \chaptername~\thechapter~– \fi\fi #1}{}}
\fancypagestyle{plain}{\fancyhf{}\renewcommand{\headrulewidth}{0pt}\fancyfoot[C]{\small\thepage}}

% ── Captions ─────────────────────────────────────────────────────────────────
\usepackage{caption}
\captionsetup{labelsep=endash, justification=centering, singlelinecheck=false,
              font=small, labelfont=bf}
\captionsetup[table]{position=above, skip=4pt}
\captionsetup[figure]{position=below, skip=6pt}

% Figures float to the nearest place that fits, top or bottom of a page, so
% that no half page is left empty; the text always cites them by number.
\usepackage{float}
\floatplacement{figure}{htbp}
\renewcommand{\topfraction}{0.85}
\renewcommand{\bottomfraction}{0.7}
\renewcommand{\textfraction}{0.12}
\renewcommand{\floatpagefraction}{0.75}
\setcounter{topnumber}{3}
\setcounter{totalnumber}{4}

% ── Tables ───────────────────────────────────────────────────────────────────
\usepackage{etoolbox}
\AtBeginEnvironment{longtable}{\small\setstretch{1.05}}
\setlength{\LTpre}{6pt}
\setlength{\LTpost}{8pt}

% ── Lists of figures and tables: "Figure 1.1 – Title" lines ──────────────────
\usepackage{tocbasic}
\DeclareTOCStyleEntry[entrynumberformat=\figureentry, numwidth=0pt]{tocline}{figure}
\DeclareTOCStyleEntry[entrynumberformat=\tableentry, numwidth=0pt]{tocline}{table}
\newcommand{\figureentry}[1]{Figure~#1~–~}
\newcommand{\tableentry}[1]{Table~#1~–~}

% ── Links ────────────────────────────────────────────────────────────────────
\AtBeginDocument{\hypersetup{hidelinks}}

% Used by latex/pin-screens.lua: screenshots are placed here (H), everything
% else floats.
\makeatletter
\newcommand{\pinfigures}{\def\fps@figure{H}}
\newcommand{\floatfigures}{\def\fps@figure{htbp}}
\makeatother
