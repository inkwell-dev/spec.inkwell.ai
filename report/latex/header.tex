% Preamble added to pandoc's LaTeX template (pandoc -H). Typesetting rules of
% the report: A4, serif body, sans-serif headings, roman page numbers for the
% front matter and arabic from the general introduction, captions centred with
% "Figure 1.1 – Title", table captions above the table and figure captions below.

% ── Fonts ────────────────────────────────────────────────────────────────────
% Liberation Serif/Sans are metric-compatible with Times New Roman and Arial.
% Glyphs they lack (✓, →, Σ, ≈…) fall back to DejaVu Sans.
\directlua{
  luaotfload.add_fallback("inkwellfallback", {
    "DejaVuSans:mode=node;",
    "DejaVuSansMono:mode=node;",
  })
}
\setmainfont{Liberation Serif}[RawFeature={fallback=inkwellfallback}]
\setsansfont{Liberation Sans}[RawFeature={fallback=inkwellfallback}]
\setmonofont{DejaVu Sans Mono}[Scale=0.85]

\usepackage{setspace}

\setlength{\parskip}{4pt plus 1pt}

% Every figure is placed where the text introduces it (H): the paragraphs that
% introduce and interpret a figure are short, so a floating figure would end
% up pages away from them. \restylefloat lets an [H] figure that does not fit
% move to the next page instead of running into the footer.
\usepackage{float}
% build.py gives every image both a width and a height cap; without this the
% image is stretched to fill both instead of fitting inside them.
\setkeys{Gin}{keepaspectratio}
% Let float manage figures, so that a pinned [H] figure that does not fit moves to
% the next page instead of running into the footer.
\restylefloat{figure}
\floatplacement{figure}{H}
\renewcommand{\topfraction}{0.9}
\renewcommand{\bottomfraction}{0.7}
\renewcommand{\textfraction}{0.12}
\renewcommand{\floatpagefraction}{0.6}
\setcounter{topnumber}{3}
\setcounter{totalnumber}{4}

% ── A3 fold-out pages (latex/foldouts.lua) ───────────────────────────────────
% One page of A3 landscape, folded into the A4 report. LuaTeX takes the size of
% each page from \pagewidth and \pageheight when it is shipped out, so the
% sizes are changed for that page only, inside a group. The margins stay those
% of the A4 pages; the extra 210 mm all go to the text width.
\usepackage{afterpage}
\newcommand{\foldoutbegin}{%
  \clearpage
  \begingroup
  \pagewidth=420mm \pageheight=297mm
  \paperwidth=420mm \paperheight=297mm
  \textwidth=\dimexpr\textwidth+210mm\relax
  \hsize=\textwidth \linewidth=\textwidth \columnwidth=\textwidth
  \headwidth=\textwidth}
\newcommand{\foldoutend}{\clearpage\endgroup}
% A tall figure turned a quarter on the sheet: the image is set rotated, and
% the box it fits into is the text area with height and width exchanged.
\newcommand{\foldoutsideways}{%
  \let\foldoutincludegraphics\includegraphics
  \renewcommand{\includegraphics}[2][]{%
    \rotatebox{90}{\foldoutincludegraphics[width=0.86\textheight,height=\textwidth,keepaspectratio]{##2}}}}

% ── Lists of figures and tables: "Figure 1.1 – Title" lines ──────────────────
\usepackage{tocbasic}
% numwidth is the box the "Figure 1.10 – " prefix sits in; the title starts
% after it. At 0pt the title was printed on top of the prefix.
\DeclareTOCStyleEntry[entrynumberformat=\figureentry, indent=0pt, numwidth=6.4em]{tocline}{figure}
\DeclareTOCStyleEntry[entrynumberformat=\tableentry, indent=0pt, numwidth=5.6em]{tocline}{table}
\newcommand{\figureentry}[1]{Figure~#1~–~}
\newcommand{\tableentry}[1]{Table~#1~–~}
